Q1: Convert a string into uppercase

Syntax / Program

#include string.h

void main() {
    char str[100];
    int i;

    printf("Enter a string: ");
    scanf("%99[^\n]", str);

    for (i = 0; str[i] != '\0'; i++) {
        if (str[i] >= 'a' && str[i] <= 'z')
            str[i] = str[i] - 32;
    }

    printf("Uppercase: %s", str);
}

Output

Uppercase: HELLO WORLD

Q3: Convert a string into sentence case

Syntax / Program

#include string.h

void main() {
    char str[100];
    int i = 0;
    int j;

    printf("Enter a string: ");
    scanf("%99[^\n]", str);

    while (str[i] == ' ')
        i++;

    for (j = 0; str[j] != '\0'; j++) {
        if (j == i) {
            if (str[j] >= 'a' && str[j] <= 'z')
                str[j] -= 32;
        } else {
            if (str[j] >= 'A' && str[j] <= 'Z')
                str[j] += 32;
        }
    }

    printf("Sentence case: %s", str);
}

Output

Enter a string: hELLO WORLD
Sentence case: Hello world
Q4: Convert a string into title case

Syntax / Program

#include string.h

void main() {
    char str[100];
    int i, newWord = 1;

    printf("Enter a string: ");
    scanf("%99[^\n]", str);

    for (i = 0; str[i] != '\0'; i++) {
        if (str[i] == ' ' || str[i] == '\t') {
            newWord = 1;
        } else {
            if (newWord && str[i] >= 'a'
                && str[i] <= 'z')
                str[i] -= 32;
            else if (!newWord && str[i] >= 'A'
                && str[i] <= 'Z')
                str[i] += 32;

            newWord = 0;
        }
    }

    printf("Title case: %s", str);
}

Output

Enter a string: hello WORLD india
Title case: Hello World India

Q13: Print the difference between corresponding characters of two strings

Syntax / Program

#include string.h


void main() {
    char a[100], b[100];
    int i, len1, len2, max;

    printf("Enter first string: ");
    scanf("%99s", a);

    printf("Enter second string: ");
    scanf("%99s", b);

    len1 = strlen(a);
    len2 = strlen(b);
    max = len1 > len2 ? len1 : len2;

    for (i = 0; i < max; i++) {
        if (i < len1 && i < len2)
            printf("%c - %c = %d\n", a[i], b[i],
                   (unsigned char)a[i] -
                   (unsigned char)b[i]);
        else
            printf("No matching character at index %d\n", i);
    }
}

Output

Enter first string: abc
Enter second string: abd
a - a = 0
b - b = 0
c - d = -1

Q18: Check palindrome using reverse

    char str[100], rev[100];

    printf("Enter a string: ");
    scanf("%99s", str);

}
