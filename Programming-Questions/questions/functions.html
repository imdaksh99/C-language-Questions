Q1: Perform addition, subtraction, multiplication and division using functions

C Program

#include stdio.h

void add() {
    int a, b;

    printf("Enter two numbers for addition: ");
    scanf("%d%d", &a, &b);

    printf("Sum = %d\n", a + b);
}

void subtract() {
    int a, b;

    printf("Enter two numbers for subtraction: ");
    scanf("%d%d", &a, &b);

    printf("Difference = %d\n", a - b);
}

void multiply() {
    int a, b;

    printf("Enter two numbers for multiplication: ");
    scanf("%d%d", &a, &b);

    printf("Product = %d\n", a * b);
}

void divide() {
    float a, b;

    printf("Enter two numbers for division: ");
    scanf("%f%f", &a, &b);

    if (b != 0)
        printf("Division = %.2f\n", a / b);
    else
        printf("Division by zero is not allowed\n");
}

void main() {
    add();
    subtract();
    add();
    multiply();
    subtract();
    multiply();
    divide();
    divide();
    add();
}

Q4: Factorial, Reverse, Palindrome, Prime, Armstrong and Fibonacci using local and global variables

C Program

#include stdio.h

int globalNumber = 5;

void factorial() {
    int i, fact = 1;

    for (i = 1; i <= globalNumber; i++)
        fact *= i;

    printf("Factorial = %d\n", fact);
}

void reverseNumber() {
    int n = globalNumber, rev = 0;

    while (n > 0) {
        rev = rev * 10 + n % 10;
        n /= 10;
    }

    printf("Reverse = %d\n", rev);
}

void palindrome() {
    int n = globalNumber, temp = n, rev = 0;

    while (n > 0) {
        rev = rev * 10 + n % 10;
        n /= 10;
    }

    if (temp == rev)
        printf("Palindrome\n");
    else
        printf("Not palindrome\n");
}

void primeNumber() {
    int i, prime = 1;

    if (globalNumber < 2)
        prime = 0;

    for (i = 2; i < globalNumber; i++) {
        if (globalNumber % i == 0)
            prime = 0;
    }

    if (prime)
        printf("Prime number\n");
    else
        printf("Not a prime number\n");
}

void armstrong() {
    int n = globalNumber, temp = n;
    int digit, sum = 0;

    while (n > 0) {
        digit = n % 10;
        sum += digit * digit * digit;
        n /= 10;
    }

    if (sum == temp)
        printf("Armstrong number\n");
    else
        printf("Not an Armstrong number\n");
}

void fibonacci() {
    int a = 0, b = 1, c, i;

    printf("Fibonacci: ");

    for (i = 1; i <= globalNumber; i++) {
        printf("%d ", a);

        c = a + b;
        a = b;
        b = c;
    }

    printf("\n");
}

void main() {
    factorial();
    reverseNumber();
    palindrome();
    primeNumber();
    armstrong();
    fibonacci();
}

Output

Factorial = 120
Reverse = 5
Palindrome
Prime number
Not an Armstrong number
Fibonacci: 0 1 1 2 3
Q5: Swap two numbers using call by address

C Program

#include stdio.h

void swap(int *a, int *b) {
    int temp;

    temp = *a;
    *a = *b;
    *b = temp;
}

void main() {
    int x, y;

    printf("Enter two numbers: ");
    scanf("%d%d", &x, &y);

    printf("Before swap: x=%d y=%d\n", x, y);

    swap(&x, &y);

    printf("After swap: x=%d y=%d\n", x, y);
}

Output

Enter two numbers: 10 20
Before swap: x=10 y=20
After swap: x=20 y=10

Q7: Make a function to calculate the power of a number

C Program

#include stdio.h

double power(double base, int exponent) {
    double result = 1;
    int i;

    for (i = 1; i <= exponent; i++)
        result = result * base;

    return result;
}

void main() {
    double number;
    int exponent;

    printf("Enter number: ");
    scanf("%lf", &number);

    printf("Enter non-negative power: ");
    scanf("%d", &exponent);

    if (exponent < 0) {
        printf("Please enter a non-negative power");
    } else {
        printf("Result = %.2f", power(number, exponent));
    }
}

Output

Enter number: 2
Enter non-negative power: 3
Result = 8.00
Q8: Make a function to calculate the cube of a number

C Program

#include stdio.h

}

    printf("Cube = %d", cube(n));
}
