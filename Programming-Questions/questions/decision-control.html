Q25: Enter date, month and year and check whether the date is valid

C Program

#include stdio.h

void main() {
    int d, m, y, maxDays, leap;

    printf("Enter day month year: ");
    scanf("%d%d%d", &d, &m, &y);

    if (y < 1 || m < 1 || m > 12) {
        printf("Invalid date");
    } else {
        leap = (y % 400 == 0 ||
                (y % 4 == 0 && y % 100 != 0));

        if (m == 2)
            maxDays = leap ? 29 : 28;
        else if (m == 4 || m == 6 ||
                 m == 9 || m == 11)
            maxDays = 30;
        else
            maxDays = 31;

        if (d >= 1 && d <= maxDays)
            printf("Valid date");
        else
            printf("Invalid date");
    }
}

Output

Enter day month year: 29 2 2024
Valid date
Q26: Enter current date and DOB and calculate age difference

C Program

#include stdio.h

void main() {
    int cd, cm, cy, bd, bm, by;
    int days, months, years;

    printf("Enter current day month year: ");
    scanf("%d%d%d", &cd, &cm, &cy);

    printf("Enter DOB day month year: ");
    scanf("%d%d%d", &bd, &bm, &by);

    if (cy < by ||
        (cy == by && cm < bm) ||
        (cy == by && cm == bm && cd < bd)) {
        printf("Invalid dates");
    } else {
        years = cy - by;
        months = cm - bm;
        days = cd - bd;

        if (days < 0) {
            months--;

            if (cm == 1 || cm == 3 || cm == 5 ||
                cm == 7 || cm == 8 || cm == 10 ||
                cm == 12)
                days += 31;
            else if (cm == 3)
                days += 28;
            else
                days += 30;
        }

        if (months < 0) {
            years--;
            months += 12;
        }

        printf("Age: %d years, %d months, %d days",
               years, months, days);
    }
}

Output

Enter current day month year: 19 10 2026
Enter DOB day month year: 10 5 2001
Age: 25 years, 5 months, 9 days
Q27: Calculate area and perimeter of a rectangle

C Program

#include stdio.h

void main() {
    float length, breadth, area, perimeter;

    printf("Enter length: ");
    scanf("%f", &length);

    printf("Enter breadth: ");
    scanf("%f", &breadth);

    if (length < 0 || breadth < 0) {
        printf("Invalid dimensions");
    } else {
        area = length * breadth;
        perimeter = 2 * (length + breadth);

        printf("Area = %.2f\n", area);
        printf("Perimeter = %.2f", perimeter);
    }
}

Output

Enter length: 10
Enter breadth: 5
Area = 50.00
Perimeter = 30.00
