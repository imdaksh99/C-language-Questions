1. Example of Pre-Increment

C Program

#include <stdio.h>

void main()
{
    int a = 5;

    printf("Before increment = %d\n", a);
    printf("Pre-increment = %d\n", ++a);
    printf("After increment = %d\n", a);
}

Output

Before increment = 5
Pre-increment = 6
After increment = 6

5. Perform Addition, Subtraction, Multiplication, Division and Remainder

C Program

#include <stdio.h>

void main()
{
    int a, b;

    printf("Enter two numbers for addition: ");
    scanf("%d%d", &a, &b);
    printf("Addition = %d\n", a + b);

    printf("\nEnter two numbers for subtraction: ");
    scanf("%d%d", &a, &b);
    printf("Subtraction = %d\n", a - b);

    printf("\nEnter two numbers for multiplication: ");
    scanf("%d%d", &a, &b);
    printf("Multiplication = %d\n", a * b);

    printf("\nEnter two numbers for division: ");
    scanf("%d%d", &a, &b);

    if (b != 0)
        printf("Division = %.2f\n", (float)a / b);
    else
        printf("Division by zero is not allowed.\n");

    printf("\nEnter two numbers for remainder: ");
    scanf("%d%d", &a, &b);

    if (b != 0)
        printf("Remainder = %d\n", a % b);
    else
        printf("Remainder by zero is not allowed.\n");
}

Sample Output

Enter two numbers for addition: 10 5
Addition = 15

Enter two numbers for subtraction: 10 5
Subtraction = 5

Enter two numbers for multiplication: 10 5
Multiplication = 50

Enter two numbers for division: 10 4
Division = 2.50

Enter two numbers for remainder: 10 3
Remainder = 1
6. Print the Last Digit of a Number

C Program

#include <stdio.h>

void main()
{
    long long num;

    printf("Enter a number: ");
    scanf("%lld", &num);

    printf("Last digit = %lld\n", num % 10);
}

Output

Enter a number: 12345
Last digit = 5

Note: Negative numbers ke liye bhi last digit ka result negative aa sakta hai. Absolute last digit chahiye to sign handle karna padega.

7. Print the Right-Hand Side of a Decimal Point

C Program

#include <stdio.h>

void main()
{
    float num;
    int rightSide;

    printf("Enter a decimal number: ");
    scanf("%f", &num);

    rightSide = (int)(num * 10) % 10;

    if (rightSide < 0)
        rightSide = -rightSide;

    printf("First digit after decimal = %d\n", rightSide);
}

Output

Enter a decimal number: 45.67
First digit after decimal = 6

Ye program decimal point ke baad ka pehla digit print karta hai. Floating-point precision ki wajah se kuch numbers mein result alag ho sakta hai.

9. Round Off a Float Number

C Program

#include <stdio.h>

void main()
{
    float num;
    int rounded;

    printf("Enter a float number: ");
    scanf("%f", &num);

    if (num >= 0)
        rounded = (int)(num + 0.5f);
    else
        rounded = (int)(num - 0.5f);

    printf("Rounded number = %d\n", rounded);
}

Output

Enter a float number: 12.7
Rounded number = 13
10. Convert a Lowercase Character into Uppercase

C Program

#include <stdio.h>

void main()
{
    char ch;

    printf("Enter a lowercase character: ");
    scanf(" %c", &ch);

    if (ch >= 'a' && ch <= 'z')
        ch = ch - 32;

    printf("Uppercase character = %c\n", ch);
}

Output

Enter a lowercase character: r
Uppercase character = R
11. Calculate the Average of Student Marks

C Program

#include <stdio.h>

void main()
{
    int m1, m2, m3;
    float average;

    printf("Enter marks of three subjects: ");
    scanf("%d%d%d", &m1, &m2, &m3);

    average = (m1 + m2 + m3) / 3.0f;

    printf("Average marks = %.2f\n", average);
}

Output

Enter marks of three subjects: 80 75 85
Average marks = 80.00

Is example mein 3 subjects hain. Subjects ki sankhya badalni ho to program ko uske hisaab se update kar sakte ho.

12. Calculate Simple Interest

C Program

#include <stdio.h>

void main()
{
    float p, r, t, si;

    printf("Enter principal amount: ");
    scanf("%f", &p);

    printf("Enter rate of interest: ");
    scanf("%f", &r);

    printf("Enter time in years: ");
    scanf("%f", &t);

    si = (p * r * t) / 100.0f;

    printf("Simple Interest = %.2f\n", si);
}

Output

Enter principal amount: 10000
Enter rate of interest: 5
Enter time in years: 2
Simple Interest = 1000.00
