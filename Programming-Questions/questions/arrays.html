Q2: Print an array in reverse order

Syntax / C Program

void main() {
    int a[5], i;

    printf("Enter 5 values: ");
    for (i = 0; i < 5; i++)

Q10: Search for a value in an array

Syntax / C Program

#include <stdio.h>

void main() {
    int a[5], i, n, found = 0;

    printf("Enter 5 values: ");
    for (i = 0; i < 5; i++)
        scanf("%d", &a[i]);

    printf("Enter value to search: ");
    scanf("%d", &n);

        if (a[i] == n) {
            found = 1;
            break;

    }

        printf("Found");
    else
        printf("Not found");
}

Q11: Sort five array values in ascending order

Syntax / C Program

#include <stdio.h>

void main() {
    int a[5], i, j, temp;

    printf("Enter 5 values: ");
    for (i = 0; i < 5; i++)
        scanf("%d", &a[i]);

    for (i = 0; i < 4; i++) {
        for (j = 0; j < 4 - i; j++) {
            if (a[j] > a[j + 1]) {
                temp = a[j];
                a[j] = a[j + 1];
                a[j + 1] = temp;
            }
        }
    }

    printf("Ascending order: ");
    for (i = 0; i < 5; i++)
        printf("%d ", a[i]);
}

Output

Enter 5 values: 50 20 40 10 30
Ascending order: 10 20 30 40 50
Q12: Find the maximum value in an array

Syntax / C Program

#include <stdio.h>

void main() {
    int a[5], i, max;

    printf("Enter 5 values: ");
    for (i = 0; i < 5; i++)
        scanf("%d", &a[i]);

    max = a[0];

    for (i = 1; i < 5; i++) {
        if (a[i] > max)
            max = a[i];
    }

    printf("Maximum value: %d", max);
}

Output

Enter 5 values: 12 45 7 89 23
Maximum value: 89
