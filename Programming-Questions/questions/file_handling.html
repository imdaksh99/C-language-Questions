Q1. Write user-entered data to a file

Enter a filename and data from the user, then write the data into the file.

Solution

#include stdio.h

void main()
{
    FILE *fp;
    char filename[100];
    char data[500];

    printf("Enter filename: ");
    scanf("%99s", filename);

    printf("Enter data: ");
    scanf(" %499[^\n]", data);

    fp = fopen(filename, "w");

    if (fp == NULL)
    {
        printf("Unable to open file.");
        return;
    }

    fprintf(fp, "%s", data);
    fclose(fp);

    printf("Data written successfully.");
}

Output

Enter filename: notes.txt
Enter data: Hello World
Data written successfully.

File content (notes.txt):

Hello World

Q2. Write data in capital letters

Enter a filename and data, convert the data to uppercase, and write it into the file.

Solution

#include stdio.h

void main()
{
    FILE *fp;
    char filename[100];
    char data[500];
    int i;

    printf("Enter filename: ");
    scanf("%99s", filename);

    printf("Enter data: ");
    scanf(" %499[^\n]", data);

    for (i = 0; data[i] != '\0'; i++)
    {
        data[i] = toupper((unsigned char)data[i]);
    }

    fp = fopen(filename, "w");

    if (fp == NULL)
    {
        printf("Unable to open file.");
        return;
    }

    fprintf(fp, "%s", data);
    fclose(fp);

    printf("Uppercase data written successfully.");
}

Output

Enter filename: capital.txt
Enter data: Hello World
Uppercase data written successfully.

File content (capital.txt):

HELLO WORLD

Q3. Write data until the user chooses to stop

Enter a filename and keep accepting data until the user chooses to stop.

Solution

#include stdio.h

void main()
{
    FILE *fp;
    char filename[100];
    char data[500];
    char choice;

    printf("Enter filename: ");
    scanf("%99s", filename);

    fp = fopen(filename, "w");

    if (fp == NULL)
    {
        printf("Unable to open file.");
        return;
    }

    do
    {
        printf("Enter data: ");
        scanf(" %499[^\n]", data);

        fprintf(fp, "%s\n", data);

        printf("Do you want to enter more data? (y/n): ");
        scanf(" %c", &choice);

    } while (choice == 'y' || choice == 'Y');

    fclose(fp);

    printf("Data written successfully.");
}

Output

Enter filename: diary.txt
Enter data: First line
Do you want to enter more data? (y/n): y
Enter data: Second line
Do you want to enter more data? (y/n): n
Data written successfully.

File content (diary.txt):

First line
Second line

Q4. Copy one file into another

Enter source and destination filenames, then copy the source file into the destination file.

Solution

#include stdio.h

void main()
{
    FILE *source, *destination;
    char sourceName[100];
    char destinationName[100];
    int ch;

    printf("Enter source filename: ");
    scanf("%99s", sourceName);

    printf("Enter destination filename: ");
    scanf("%99s", destinationName);

    source = fopen(sourceName, "r");

    if (source == NULL)
    {
        printf("Unable to open source file.");
        return;
    }

    destination = fopen(destinationName, "w");

    if (destination == NULL)
    {
        printf("Unable to open destination file.");
        fclose(source);
        return;
    }

    while ((ch = fgetc(source)) != EOF)
    {
        fputc(ch, destination);
    }

    fclose(source);
    fclose(destination);

    printf("File copied successfully.");
}

Output

Enter source filename: notes.txt
Enter destination filename: copy.txt
File copied successfully.

File content (copy.txt):

Hello World

Q5. Rename a file

Enter a filename and data, write the data into the file, then rename the file.

Solution

#include stdio.h

void main()
{
    FILE *fp;
    char filename[100];
    char newFilename[100];
    char data[500];

    printf("Enter filename: ");
    scanf("%99s", filename);

    printf("Enter data: ");
    scanf(" %499[^\n]", data);

    fp = fopen(filename, "w");

    if (fp == NULL)
    {
        printf("Unable to open file.");
        return;
    }

    fprintf(fp, "%s", data);
    fclose(fp);

    printf("Enter new filename: ");
    scanf("%99s", newFilename);

    if (rename(filename, newFilename) == 0)
    {
        printf("File renamed successfully.");
    }
    else
    {
        printf("Unable to rename file.");
    }
}

Output

Enter filename: old.txt
Enter data: Sample text
Enter new filename: new.txt
File renamed successfully.

File content (new.txt):

Sample text

Q6. Count characters in a file

Enter a filename and count the characters present in the file.

Solution

#include stdio.h

void main()
{
    FILE *fp;
    char filename[100];
    int ch;
    int count = 0;

    printf("Enter filename: ");
    scanf("%99s", filename);

    fp = fopen(filename, "r");

    if (fp == NULL)
    {
        printf("Unable to open file.");
        return;
    }

    while ((ch = fgetc(fp)) != EOF)
    {
        count++;
    }

    fclose(fp);

    printf("Number of characters: %d", count);
}

Output

Enter filename: notes.txt
Number of characters: 11

Here, notes.txt contains Hello World, which has 11 characters including the space.

Q7. Concatenate two files into a third file

Enter two source filenames and a third filename, then combine the contents of the first two files into the third file.

Solution

#include stdio.h

void main()
{
    FILE *file1, *file2, *file3;
    char name1[100];
    char name2[100];
    char name3[100];
    int ch;

    printf("Enter first filename: ");
    scanf("%99s", name1);

    printf("Enter second filename: ");
    scanf("%99s", name2);

    printf("Enter destination filename: ");
    scanf("%99s", name3);

    file1 = fopen(name1, "r");

    if (file1 == NULL)
    {
        printf("Unable to open first file.");
        return;
    }

    file2 = fopen(name2, "r");

    if (file2 == NULL)
    {
        printf("Unable to open second file.");
        fclose(file1);
        return;
    }

    file3 = fopen(name3, "w");

    if (file3 == NULL)
    {
        printf("Unable to open destination file.");
        fclose(file1);
        fclose(file2);
        return;
    }

    while ((ch = fgetc(file1)) != EOF)
    {
        fputc(ch, file3);
    }

    while ((ch = fgetc(file2)) != EOF)
    {
        fputc(ch, file3);
    }

    fclose(file1);
    fclose(file2);
    fclose(file3);

    printf("Files concatenated successfully.");
}

Output

Enter first filename: first.txt
Enter second filename: second.txt
Enter destination filename: combined.txt
Files concatenated successfully.

File content (first.txt):

Hello

File content (second.txt):

World

File content (combined.txt):

Hello World
