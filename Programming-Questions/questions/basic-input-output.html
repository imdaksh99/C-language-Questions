Q3: Read only 4 characters and display each on a separate line

O

Q6: Print your name with blinking effect 5 times

Syntax / C Program (Turbo C)

#include stdio.h
#include conio.h
#include dos.h

void main() {
    int i;

    clrscr();

    for (i = 0; i < 5; i++) {
        gotoxy(35, 12);
        textcolor(WHITE);
        cprintf("Rahul");
        delay(500);

        gotoxy(35, 12);
        cprintf("     ");
        delay(500);
    }

    gotoxy(35, 14);
    printf("Blinking completed");
    getch();
}

Output

Rahul (blinks 5 times)

Blinking completed

Note: Turbo C ke conio.h aur dos.h ki zaroorat hai. Modern compilers mein ye functions available nahi hote.

Q12: Enter password and display it as asterisks

Syntax / C Program (Turbo C)

#include stdio.h
#include conio.h

void main() {
    char password[20];
    char ch;
    int i = 0;

    printf("Enter password: ");

    while (i < 19) {
        ch = getch();

        if (ch == '\r')
            break;

        if (ch == '\b') {
            if (i > 0) {
                i--;
                printf("\b \b");
            }
        } else {
            password[i++] = ch;
            printf("*");
        }
    }

    password[i] = '\0';

    printf("\nPassword entered: %s", password);
    getch();
}

Output

Enter password: ******
Password entered: secret

Note: Password ko dobara print karne wali line sirf demonstration ke liye hai. Real password input mein ise print nahi karna chahiye.

Q13: Enter two characters and print their ASCII values

Syntax / C Program

#include stdio.h

void main() {
    char ch1, ch2;

    printf("Enter first character: ");
    scanf(" %c", &ch1);

    printf("Enter second character: ");
    scanf(" %c", &ch2);

    printf("ASCII value of %c = %d\n", ch1,
           (unsigned char)ch1);
    printf("ASCII value of %c = %d\n", ch2,
           (unsigned char)ch2);
}

Output

Enter first character: A
Enter second character: B
ASCII value of A = 65
ASCII value of B = 66
Q14: Print ASCII characters from 0 to 127

Syntax / C Program

#include stdio.h

void main() {
    int i;

    for (i = 0; i <= 127; i++) {
        printf("%3d : ", i);

        if (i < 32 || i == 127)
            printf("Non-printable/control");
        else
            printf("%c", i);

        printf("\n");
    }
}

Output (sample)

  0 : Non-printable/control
  1 : Non-printable/control
  2 : Non-printable/control
...
 65 : A
 66 : B
 67 : C
...
127 : Non-printable/control

Note: ASCII values 0–31 and 127 are control characters; they are not all printable symbols.
